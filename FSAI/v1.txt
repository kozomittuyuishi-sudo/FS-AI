async function getBackendStatus() {
  try {
    const res = await fetch('http://127.0.0.1:8000/', { cache: 'no-store' });
    if (!res.ok) return { message: 'Backend unreachable' };
    return res.json();
  } catch (error) {
    return { message: 'Connection failed' };
  }
}

export default async function Home() {
  const data = await getBackendStatus();

  return (
    <main className="flex min-h-screen flex-col items-center justify-center p-24 bg-gray-900 text-white">
      <div className="z-10 max-w-5xl w-full items-center justify-between font-mono text-sm">
        <h1 className="text-4xl font-bold mb-4">FashionSense AI</h1>
        <p className="text-lg">Backend Status: <span className="text-green-400">{data.message}</span></p>
      </div>
    </main>
  );
}
